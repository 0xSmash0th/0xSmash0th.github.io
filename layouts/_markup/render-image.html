{{- /* Obsidian resolves image paths from the vault root, so notes reference /static/foo.png. Hugo publishes the *contents* of static/ at the site root, where that path does not exist. Strip the prefix at render time so one path works in both tools and the markdown stays Obsidian-native. */ -}} {{- $dest := .Destination -}} {{- if hasPrefix $dest "/static/" -}} {{- $dest = strings.TrimPrefix "/static" $dest -}} {{- end -}}
